The shape, in words: a WITH clause named new_vote inserts the vote (report_id, account_id, kind, value, geohash_cell from the five args, created_at from the column default) and returns report_id, account_id, kind, value, geohash_cell and created_at; a second WITH clause named extended runs UPDATE reports r SET expires_at = GREATEST(r.expires_at, LEAST(r.created_at plus sqlc.arg(cap_seconds)::bigint times interval '1 second', nv.created_at plus sqlc.arg(life_seconds)::bigint times interval '1 second')) FROM new_vote nv WHERE r.id = nv.report_id AND sqlc.arg(extend)::boolean AND r.expires_at > nv.created_at AND NOT EXISTS (SELECT 1 FROM votes v WHERE v.report_id = nv.report_id AND v.kind = nv.kind AND v.value = nv.value AND (v.geohash_cell = nv.geohash_cell OR v.account_id = nv.account_id) AND v.account_id IS DISTINCT FROM sqlc.narg(reporter_account_id)::bigint) RETURNING r.expires_at; the main SELECT is COALESCE((SELECT e.expires_at FROM extended e), r.expires_at)::timestamptz AS expires_at FROM reports r WHERE r.id = sqlc.arg(report_id).
